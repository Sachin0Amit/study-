    52→/** Determine the most appropriate physics/math/chemistry diagram from question text & explanation */
    53→export function detectDiagramTopic(
    54→  text: string,
    55→  explanation?: string,
    56→  chapterId?: string
    57→): DiagramTopic {
    58→  const combined = `${text} ${explanation ?? ''} ${chapterId ?? ''}`.toLowerCase()
    59→
    60→  // --- Chemistry Topics ---
    61→  if (
    62→    /titrat|equivalence point|buffer|ph |poh|henderson|acid.*base|indicator|neutraliz|weak acid|pka|pkb|hydrolysis|chem-ch1\b|chem-ch5\b|chem-ch27\b|le chatelier|equilibrium constant|q vs k|haber process|reaction quotient/i.test(
    63→      combined
    64→    )
    65→  ) {
    66→    return 'titration_equilibrium'
    67→  }
    68→  if (
    69→    /galvanic|voltaic|daniell cell|anode|cathode|nernst|electrode potential|standard reduction|salt bridge|faraday|electrochemical|chem-ch6\b/i.test(
    70→      combined
    71→    )
    72→  ) {
    73→    return 'electrochem_galvanic'
    74→  }
    75→  if (
    76→    /rate law|order of reaction|zero order|first order|second order|half-life|arrhenius|activation energy|catalyst.*barrier|collision theory|chem-ch7\b|chem-ch19\b|chem-ch20\b|chem-ch26\b|chem-ch28\b/i.test(
    77→      combined
    78→    )
    79→  ) {
    80→    return 'reaction_kinetics'
    81→  }
    82→  if (
    83→    /periodic trend|electronegativ|atomic radi|ionization energy|shielding effect|effective nuclear charge|electron affinity|mendeleev|chem-ch11-s1|chem-ch13\b|chem-ch14\b|chem-ch31\b/i.test(
    84→      combined
    85→    )
    86→  ) {
    87→    return 'periodic_trends'
    88→  }
    89→  if (
    90→    /vsepr|molecular geometry|bond angle|lone pair|tetrahedral|trigonal|octahedral|bent molecule|sp3|sp2|sp\b|hybridiz|dipole moment|chem-ch8\b|chem-ch9\b|chem-ch10\b|chem-ch11\b|chem-ch12\b|chem-ch17\b|chem-ch21\b|chem-ch22\b|chem-ch23\b|chem-ch25\b|chem-ch32\b/i.test(
    91→      combined
    92→    )
    93→  ) {
    94→    return 'molecular_vsepr'
    95→  }
    96→
    97→  // --- Physics & Math Topics ---
    98→  if (
    99→    /balmer|lyman|paschen|bohr|hydrogen|ground state|13\.6|spectral|rydberg|energy level|ionization|h-alpha|coletta-ch29|ch29|quantum atom|chem-ch2\b|chem-ch15\b|chem-ch16\b/i.test(
   100→      combined
   101→    )
   102→  ) {
   103→    return 'bohr_transitions'
   104→  }
   105→  if (
   106→    /projectile|trajectory|launch|angle.*velocity|range.*gravity|parabola|horizontal velocity|vertical velocity|flight time|maximum height|coletta-ch3|young-ch3/i.test(
   107→      combined
   108→    )
   109→  ) {
   110→    return 'projectile'
   111→  }
   112→  if (
   113→    /centripetal|circular motion|orbit.*speed|radius.*angular|tangential velocity|uniform circular/i.test(
   114→      combined
   115→    )
   116→  ) {
   117→    return 'circular_motion'
   118→  }
   119→  if (
   120→    /incline|friction|normal force|newton|free body|tension.*pulley|slope|coletta-ch4|young-ch5/i.test(
   121→      combined
   122→    )
   123→  ) {
   124→    return 'free_body'

   127→    /interference|double slit|diffraction|standing wave|node.*antinode|wavelength.*fringe|crest.*trough|superposition|slit.*separation|coletta-ch24|young-ch35/i.test(
   128→      combined
   129→    )
   130→  ) {
   131→    return 'wave_interference'
   132→  }
   133→  if (
   134→    /doppler|frequency.*approaching|source.*moving|sound.*pitch|redshift|blueshift|speed of sound|mach.*number|coletta-ch16|young-ch16/i.test(
   135→      combined
   136→    )
   137→  ) {
   138→    return 'doppler'
   139→  }
   140→  if (
   141→    /coulomb|electric field|charge.*force|point charge|gauss.*law|potential difference|equipotential|coletta-ch17|young-ch21/i.test(
   142→      combined
   143→    )
   144→  ) {

   146→  }

   148→    /resistor|ohm|kirchhoff|voltage|series.*parallel|current.*ampere|circuit|equivalent resistance|elec-|coletta-ch19|young-ch25|young-ch26/i.test(

   156→      combined
   157→    )
   158→  ) {
   159→    return 'optics_snell'

   162→    /isothermal|adiabatic|isobaric|isochoric|pv diagram|carnot|heat engine|ideal gas|work done.*gas|coletta-ch15|young-ch19|chem-ch3\b|chem-ch4\b|chem-ch30\b/i.test(
   163→      combined

   165→  ) {

   167→  }
   168→  if (
   169→    /derivative|tangent line|riemann|integral.*area|rate of change|limit as|secant|dy\/dx|calc-|stewart/i.test(
   170→      combined
   171→    )
   172→  ) {
   173→    return 'calculus_tangent'

   182→
   183→  // Safe domain-aware fallbacks before universal physics
   184→  if (/chem-|chemistry|reaction|molecule|molar|element/i.test(combined)) {
   185→    return 'molecular_vsepr'
   186→  }
   187→
   188→  return 'universal_physics'
   189→}

   502→/*  2. Projectile Motion Parabolic Trajectory Diagram                  */
   503→/* ------------------------------------------------------------------ */
   504→
   505→export function ProjectileMotionDiagram({ playing }: { playing: boolean }) {
   506→  const [angle, setAngle] = useState(45)
   507→  const [v0, setV0] = useState(30)
   508→  const g = 9.8
   509→
   510→  const [t, setT] = useState(0)
   511→  useEffect(() => {
   512→    if (!playing) return
   513→    const interval = setInterval(() => {
   514→      setT((prev) => (prev + 1) % 100)
   515→    }, 40)
   516→    return () => clearInterval(interval)
   517→  }, [playing])
   518→
   519→  const rad = (angle * Math.PI) / 180
   520→  const v0x = v0 * Math.cos(rad)
   521→  const v0y = v0 * Math.sin(rad)
   522→  const flightTime = (2 * v0y) / g
   523→  const maxH = (v0y * v0y) / (2 * g)
   524→  const range = (v0 * v0 * Math.sin(2 * rad)) / g
   525→
   526→  // Scaled coordinates for SVG (width 600, height 240)
   527→  const scaleX = 500 / Math.max(1, range)
   528→  const scaleY = 160 / Math.max(1, maxH)
   529→
   530→  const curT = (t / 100) * flightTime
   531→  const curX = v0x * curT
   532→  const curY = v0y * curT - 0.5 * g * curT * curT
   533→  const curVx = v0x
   534→  const curVy = v0y - g * curT
