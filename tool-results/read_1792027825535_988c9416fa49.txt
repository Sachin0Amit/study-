    50→export interface Subject {
    51→  id: string
    52→  name: string
    53→  slug: string
    54→  description: string
    55→  icon: string // lucide icon name key
    56→  accent: string // hex accent color
    57→  tagline: string
    58→  group?: string // homepage section (Foundations, Computing Systems, ...)
    59→}
    60→
    61→export interface BookMeta {
    62→  id: string
    63→  subjectId: string
    64→  title: string
    65→  author: string
    66→  edition?: string
    67→  year?: string
    68→  description: string
    69→  pages?: number
    70→  accent: string // hex accent for cards
    71→  sources?: string[] // for master volumes: the books merged into this one
    72→}
    73→
    74→export interface ChapterMeta {
    75→  id: string // e.g. "rosen-ch1"
    76→  bookId: string
    77→  number: number
    78→  title: string
    79→  shortTitle: string
    80→  description: string
    81→  icon: string // lucide icon key in chapterIcons map
    82→  pages: string // e.g. "24–143"
    83→  interactiveKey: string // key into interactives registry
    84→}
    85→
    86→export interface SectionMeta {
    87→  id: string // e.g. "rosen-ch1-s1"
    88→  chapterId: string
    89→  number: number // section number, e.g. 1.1
    90→  title: string
    91→  description: string
    92→}
    93→
    94→// ---------- Master content for a section ----------
    95→
    96→export interface Concept {
    97→  term: string
    98→  kind: 'definition' | 'theorem' | 'law' | 'method' | 'notation' | 'insight'
    99→  statement: string // supports simple **bold** markdown-ish
   100→  detail?: string
   101→}
   102→
   103→export interface WorkedExample {
   104→  title: string
   105→  problem: string
   106→  steps: { label: string; text: string }[]
   107→  result?: string
   108→  diagram?: string // visual physics diagram or illustration
   109→}
   110→
   111→export interface QuizQuestion {
   112→  q: string
   113→  options: string[]
   114→  answer: number // index into options
   115→  explanation: string
   116→  diagram?: string // visual physics diagram or illustration solving the problem
   117→}
   118→
   119→/* ------------------------------------------------------------------ */
   120→/*  The depth ladder — "from basics to a PhD"                          */
   121→/*  The section core above IS the undergraduate material (Level I).   */
   122→/*  Optional tiers then carry the very same topics to graduate         */
   123→/*  coursework level (masters) and on to the research frontier (phd).  */
   124→/* ------------------------------------------------------------------ */
   125→
   126→export interface DepthBlock {
   127→  kind: 'para' | 'theorem' | 'proof' | 'derivation' | 'remark' | 'example' | 'application' | 'open'
   128→  heading?: string // required for example, recommended for open
   129→  body: string // flowing prose; **bold**, $...$/$$...$$ KaTeX, `code`
   130→}
   131→
   132→export interface DepthTier {
   133→  tier: 'masters' | 'phd'
   134→  title: string
   135→  blocks: DepthBlock[]
   136→}
   137→
   138→/** A textbook figure — an authored inline SVG set into the reading flow,
   139→ *  like a figure plate in a printed book. `svg` is static authored data
   140→ *  (never user input): responsive via viewBox, strokes in currentColor so
   141→ *  it adapts to light/dark paper. */
   142→export interface LessonFigure {
   143→  caption: string // rendered as "Figure N — caption"; supports $...$ math
   144→  svg: string // inline SVG markup (viewBox based, width 100%)
   145→  sketchImage?: string // authentic hand-drawn scientific notebook sketch image path, e.g. "/diagrams/bohr_atom_sketch.jpg"
   146→  sketchCaption?: string // handwritten notes / summary caption for sketch view
   147→  afterConcept?: number // 1-based concept ordinal to follow; omit → after all concepts
   148→}
   149→
   150→export interface DerivationStep {
   151→  label: string // e.g. "Step 1: Momentum transfer per collision"
   152→  formula?: string // KaTeX formula, e.g. "$$\\Delta p_x = 2mv_x$$"
   153→  explanation: string // pedagogical rationale for this step
   154→}
   155→
   156→export interface DetailedDerivation {
   157→  title: string
   158→  statement: string // target formula or law to derive, e.g. "Derivation of Kinetic Pressure $P = \\frac{1}{3}nm\\langle v^2\\rangle$"
   159→  prerequisites?: string[] // what physics/math is assumed for this derivation
   160→  steps: DerivationStep[]
   161→  result: string // final derived formula and interpretation
   162→}
   163→
   164→export interface SectionContent {
   165→  sectionId: string
   166→  readingTime: number // minutes
   167→  prerequisites?: string[] // concepts or prior sections required before this lesson
   168→  intro: string
   169→  objectives: string[]
   170→  concepts: Concept[]
   171→  figures?: LessonFigure[]
   172→  derivations?: DetailedDerivation[]
   173→  examples: WorkedExample[]
   174→  keyTakeaways: string[]
   175→  quiz: (QuizQuestion | undefined)[]
   176→  depth?: DepthTier[]
   177→}
